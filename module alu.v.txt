module alu (
    input  [7:0] A, B,       // 8-bit inputs
    input  [2:0] OpCode,     // 3-bit Operation Selector (8 operations)
    output reg [7:0] ALU_Out,// 8-bit ALU output
    output reg CarryOut      // Carry flag for addition
);

    always @(*) begin
        // Default values to prevent latches
        ALU_Out  = 8'b0;
        CarryOut = 1'b0;
        
        case (OpCode)
            3'b000: {CarryOut, ALU_Out} = A + B;  // Addition
            3'b001: ALU_Out = A - B;              // Subtraction
            3'b010: ALU_Out = A & B;              // Bitwise AND
            3'b011: ALU_Out = A | B;              // Bitwise OR
            3'b100: ALU_Out = A ^ B;              // Bitwise XOR
            3'b101: ALU_Out = ~(A | B);           // Bitwise NOR
            3'b110: ALU_Out = A << 1;             // Logical Shift Left
            3'b111: ALU_Out = A >> 1;             // Logical Shift Right
            default: ALU_Out = 8'b0;
        endcase
    end
endmodule
